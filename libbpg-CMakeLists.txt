# Minimal CMakeLists.txt for libbpg - not part of upstream libbpg (which
# ships a hand-written Makefile targeting native/legacy-emscripten
# builds only). This builds *only* the decoder side (libbpg.c + the
# stripped-down libavcodec/libavutil HEVC decoder it bundles), matching
# exactly the LIBBPG_OBJS file list from libbpg's own Makefile. The
# x265/JCTVC-based encoder is intentionally excluded - this repo's
# libbpg filter is decode-only.
cmake_minimum_required(VERSION 3.7)
project(libbpgdec C)

set(CMAKE_C_STANDARD 99)

add_definitions(
	-D_ISOC99_SOURCE
	-D_POSIX_C_SOURCE=200112
	-D_XOPEN_SOURCE=600
	-DHAVE_AV_CONFIG_H
	-D_GNU_SOURCE=1
	-DUSE_VAR_BIT_DEPTH
	-DUSE_PRED
	-DCONFIG_BPG_VERSION="1.0"
	-DEMSCRIPTEN
	-fpic
)

include_directories(${CMAKE_CURRENT_SOURCE_DIR})

set(LIBBPGDEC_SRC
	libavcodec/hevc_cabac.c
	libavcodec/hevc_filter.c
	libavcodec/hevc.c
	libavcodec/hevcpred.c
	libavcodec/hevc_refs.c
	libavcodec/hevcdsp.c
	libavcodec/hevc_mvs.c
	libavcodec/hevc_ps.c
	libavcodec/hevc_sei.c
	libavcodec/utils.c
	libavcodec/cabac.c
	libavcodec/golomb.c
	libavcodec/videodsp.c
	libavutil/mem.c
	libavutil/buffer.c
	libavutil/log2_tab.c
	libavutil/frame.c
	libavutil/pixdesc.c
	libavutil/md5.c
	libbpg.c
)

add_library(bpgdec STATIC ${LIBBPGDEC_SRC})
